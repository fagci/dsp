# Нативный слой приложения: whisper.cpp (речь → текст) в виде JNI-библиотеки.
# Две сборки из одних исходников (build.sh): dsp_whisper — CPU (arm64 с dotprod + fp16), dsp_whisper_cl — то же и GPU через OpenCL (Adreno).
# Отдельная библиотека для OpenCL нужна потому, что libOpenCL.so попадает в NEEDED: на телефоне без неё библиотека не загрузилась бы вовсе.
# Какую грузить — решает Kotlin (WhisperNative.kt) по /proc/cpuinfo и по наличию libOpenCL.so.
cmake_minimum_required(VERSION 3.22)
project(dsp_ai C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(DSP_OPENCL "GPU через OpenCL (нужны заголовки и libOpenCL.so: OpenCL_INCLUDE_DIR, OpenCL_LIBRARY)" OFF)
set(DSP_LIB "dsp_whisper" CACHE STRING "имя библиотеки")
set(DSP_WHISPER_TAG "v1.9.2" CACHE STRING "тег whisper.cpp")
set(DSP_CPU_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "-march для ядер ggml")

include(FetchContent)
FetchContent_Declare(whisper_src GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG ${DSP_WHISPER_TAG} GIT_SHALLOW TRUE)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)                 # в NDK libomp отдельной библиотекой — лишняя зависимость
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)                 # собираем под телефон, а не под машину сборки
set(GGML_CPU_ARM_ARCH "${DSP_CPU_ARCH}" CACHE STRING "" FORCE)
if(DSP_OPENCL)
  set(GGML_OPENCL ON CACHE BOOL "" FORCE)
  set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
  set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
endif()
FetchContent_MakeAvailable(whisper_src)

add_library(${DSP_LIB} SHARED whisper_jni.cpp)
target_link_libraries(${DSP_LIB} PRIVATE whisper ggml log android)
target_compile_options(${DSP_LIB} PRIVATE -O3 -fvisibility=hidden -fvisibility-inlines-hidden -ffunction-sections -fdata-sections)
target_link_options(${DSP_LIB} PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
